4.3 核心代码（InstrumentProxy）

import asyncio
import logging
import time
import uuid
from multiprocessing import Queue
from threading import Lock
from typing import Dict, Any, Optional

logger = logging.getLogger(__name__)


class InstrumentProxy:
    """仪器代理进程核心类，所有仪器操作的唯一入口。"""

    def __init__(
        self,
        request_queue: Queue,
        response_queue: Queue,
        config: Dict[str, Any],
        simulation: bool = False
    ):
        self.request_queue = request_queue
        self.response_queue = response_queue
        self.config = config
        self.simulation = simulation

        self._instruments: Dict[str, Any] = {}
        self._locks: Dict[str, Lock] = {}
        self._call_log: list = []
        self._running = False
        self._watchdog_task: Optional[asyncio.Task] = None
        self._last_heartbeat = time.time()

    def run(self):
        """代理进程主循环，处理请求并启动看门狗协程。"""
        self._running = True
        self._init_instruments()
        loop = asyncio.new_event_loop()
        asyncio.set_event_loop(loop)
        self._watchdog_task = loop.create_task(self._watchdog())
        logger.info("InstrumentProxy started, simulation=%s", self.simulation)

        try:
            while self._running:
                try:
                    msg = self.request_queue.get(timeout=1.0)
                    if msg is None:  # 毒丸消息，退出
                        break
                    self._handle_request(msg)
                except Exception:
                    pass  # 超时继续循环
        finally:
            self._running = False
            self._flush_log()
            loop.cancel(self._watchdog_task)
            loop.close()

    def _init_instruments(self):
        """根据配置初始化仪器（真实或Mock）。"""
        from ate_platform.drivers.mock.mock_factory import MockDriverFactory
        from ate_platform.drivers.registry import DriverRegistry

        for inst_id, inst_cfg in self.config.get("instruments", {}).items():
            if self.simulation:
                driver = MockDriverFactory.get(inst_cfg["type"], inst_id)
            else:
                driver = DriverRegistry.create(inst_cfg["type"], inst_cfg)
            self._instruments[inst_id] = driver
            self._locks[inst_id] = Lock()
            logger.info("Initialized instrument: %s (sim=%s)", inst_id, self.simulation)

    def _handle_request(self, msg: Dict[str, Any]):
        """处理请求，含故障注入、per-instrument锁、调用日志。"""
        req_id = msg.get("req_id", str(uuid.uuid4()))
        inst_id = msg["instrument_id"]
        method = msg["method"]
        args = msg.get("args", ())
        kwargs = msg.get("kwargs", {})

        lock = self._locks.get(inst_id)
        if lock is None:
            self._response(req_id, success=False, error=f"Unknown instrument: {inst_id}")
            return

        with lock:
            start = time.time()
            try:
                # 故障注入检查
                inst = self._instruments[inst_id]
                if hasattr(inst, "_check_fault"):
                    inst._check_fault(method)

                result = getattr(inst, method)(*args, **kwargs)
                elapsed = time.time() - start
                self._log_call(inst_id, method, args, kwargs, result, elapsed)
                self._response(req_id, success=True, result=result)
            except Exception as e:
                elapsed = time.time() - start
                self._log_call(inst_id, method, args, kwargs, str(e), elapsed, error=True)
                self._response(req_id, success=False, error=str(e))

    def _response(self, req_id: str, success: bool, result=None, error=None):
        """发送响应到响应队列。"""
        self.response_queue.put({
            "req_id": req_id,
            "success": success,
            "result": result,
            "error": error,
            "timestamp": time.time()
        })

    def _log_call(self, inst_id, method, args, kwargs, result, elapsed, error=False):
        """记录调用日志。"""
        self._call_log.append({
            "inst_id": inst_id,
            "method": method,
            "args": args,
            "kwargs": kwargs,
            "result": result,
            "elapsed_ms": round(elapsed * 1000, 2),
            "error": error,
            "timestamp": time.time()
        })
        # 每100条刷盘一次
        if len(self._call_log) >= 100:
            self._flush_log()

    def _flush_log(self):
        """刷盘调用日志。"""
        if not self._call_log:
            return
        log_path = self.config.get("call_log_path", "/tmp/proxy_calls.jsonl")
        import json
        with open(log_path, "a") as f:
            for entry in self._call_log:
                f.write(json.dumps(entry) + "\n")
        self._call_log.clear()

    async def _watchdog(self):
        """看门狗协程，心跳检测，崩溃自动重启。"""
        while self._running:
            await asyncio.sleep(5.0)
            now = time.time()
            if now - self._last_heartbeat > 30.0:
                logger.critical("InstrumentProxy heartbeat lost, restarting...")
                self._init_instruments()
                self._last_heartbeat = now
            else:
                self._last_heartbeat = now

4.4 仪器客户端（InstrumentClient）

import time
import uuid
import queue
from multiprocessing import Queue
from typing import Any, Dict


class InstrumentClient:
    """执行进程侧的仪器客户端，通过IPC与代理进程通信。"""

    def __init__(
        self,
        request_queue: Queue,
        response_queue: Queue,
        resource_id: str,
        timeout: float = 10.0
    ):
        self.request_queue = request_queue
        self.response_queue = response_queue
        self.resource_id = resource_id
        self.timeout = timeout
        self._pending: Dict[str, float] = {}

    def _call(self, method: str, *args, **kwargs) -> Any:
        """发送请求，等待响应，含超时处理。"""
        req_id = str(uuid.uuid4())
        self.request_queue.put({
            "req_id": req_id,
            "instrument_id": self.resource_id,
            "method": method,
            "args": args,
            "kwargs": kwargs
        })
        self._pending[req_id] = time.time()

        deadline = time.time() + self.timeout
        while time.time() < deadline:
            try:
                resp = self.response_queue.get(timeout=0.1)
                if resp["req_id"] == req_id:
                    self._pending.pop(req_id, None)
                    if resp["success"]:
                        return resp["result"]
                    raise RuntimeError(resp["error"])
                else:
                    # 非本请求的响应，放回队列
                    self.response_queue.put(resp)
            except queue.Empty:
                continue

        self._pending.pop(req_id, None)
        raise TimeoutError(f"Instrument call timed out: {self.resource_id}.{method}")

    def write(self, cmd: str):
        return self._call("write", cmd)

    def ask(self, cmd: str) -> str:
        return self._call("ask", cmd)

    def call_method(self, name: str, *args, **kwargs):
        return self._call(name, *args, **kwargs)

    def __getattr__(self, name: str):
        if name.startswith("_"):
            raise AttributeError(name)
        def proxy(*args, **kwargs):
            return self._call(name, *args, **kwargs)
        return proxy

import logging
from pymeasure.instruments import Instrument

logger = logging.getLogger(__name__)

class PlatformSCPIInstrument(Instrument):
    """SCPI设备统一基类，扩展pymeasure.Instrument以支持平台级特性"""

    def __init__(self, resource_id, adapter=None, simulation=False, **kwargs):
        """
        初始化SCPI设备
        :param resource_id: VISA资源字符串或TCP地址
        :param adapter: pymeasure Adapter实例，为None时自动根据resource_id推断
        :param simulation: 是否启用仿真模式
        """
        super().__init__(adapter or resource_id, "Platform SCPI Device", **kwargs)
        self.resource_id = resource_id
        self._simulation_mode = simulation
        if simulation:
            logger.info(f"[SIM] SCPI设备初始化(仿真模式): {resource_id}")

    @property
    def simulation_mode(self) -> bool:
        """当前是否处于仿真模式"""
        return self._simulation_mode

    def identify(self) -> str:
        """获取设备标识 (*IDN?)"""
        if self._simulation_mode:
            return "SIM,PlatformSCPI,0,1.0"
        return self.ask("*IDN?").strip()

    def reset(self):
        """复位设备 (*RST)"""
        if self._simulation_mode:
            logger.debug("[SIM] 设备已复位")
            return
        self.write("*RST")

    def self_test(self) -> bool:
        """执行设备自检 (*TST?)，返回True表示通过"""
        if self._simulation_mode:
            return True
        result = self.ask("*TST?").strip()
        return result == "0"

# src/ate_platform/drivers/base/tcp_instrument.py
import socket
import logging
import struct

logger = logging.getLogger(__name__)

class PlatformTCPInstrument:
    """自定义TCP协议设备基类"""

    def __init__(self, resource_id, host, port, timeout=5.0, simulation=False):
        self.resource_id = resource_id
        self.host = host
        self.port = port
        self.timeout = timeout
        self._simulation_mode = simulation
        self._socket: socket.socket | None = None

    @property
    def simulation_mode(self) -> bool:
        return self._simulation_mode

    def connect(self):
        if self._simulation_mode:
            logger.info(f"[SIM] TCP连接(仿真): {self.host}:{self.port}")
            return
        self._socket = socket.socket(socket.AF_INET, socket.SOCK_STREAM)
        self._socket.settimeout(self.timeout)
        self._socket.connect((self.host, self.port))
        logger.info(f"TCP已连接: {self.host}:{self.port}")

    def disconnect(self):
        if self._socket:
            self._socket.close()
            self._socket = None
            logger.info(f"TCP已断开: {self.host}:{self.port}")

    def send(self, data: bytes):
        if self._simulation_mode:
            return
        if not self._socket:
            raise ConnectionError("TCP未连接")
        self._socket.sendall(data)

    def recv(self, bufsize: int = 4096) -> bytes:
        if self._simulation_mode:
            return b""
        if not self._socket:
            raise ConnectionError("TCP未连接")
        return self._socket.recv(bufsize)

    def _recv_exact(self, n: int) -> bytes:
        """精确接收n字节"""
        if self._simulation_mode:
            return b"\x00" * n
        buf = bytearray()
        while len(buf) < n:
            chunk = self._socket.recv(n - len(buf))
            if not chunk:
                raise ConnectionError("TCP连接中断")
            buf.extend(chunk)
        return bytes(buf)

    def send_and_recv(self, data: bytes, recv_size: int = 4096) -> bytes:
        """发送并接收响应"""
        self.send(data)
        return self.recv(recv_size)

    def identify(self) -> str:
        if self._simulation_mode:
            return "SIM,TCPDevice,0,1.0"
        raise NotImplementedError("子类需实现identify协议")

    def reset(self):
        if self._simulation_mode:
            logger.debug("[SIM] TCP设备已复位")
            return
        raise NotImplementedError("子类需实现reset协议")

5.5 TCP设备驱动示例（Chroma电子负载）

基于Chroma私有协议帧格式实现电子负载驱动。

# src/ate_platform/drivers/tcp/chroma_eload.py
import struct
import logging
from ate_platform.drivers.base.tcp_instrument import PlatformTCPInstrument

logger = logging.getLogger(__name__)

class Chroma63004(PlatformTCPInstrument):
    """Chroma 63000系列电子负载TCP驱动"""
    
    HEAD = 0xAA55
    ADDR = 0x01
    
    # 命令字定义
    CMD_SET_MODE = 0x20
    CMD_SET_CURRENT = 0x21
    CMD_MEASURE = 0x30
    CMD_IDN = 0x01
    CMD_RESET = 0x02

    def _build_frame(self, cmd: int, data: bytes = b"") -> bytes:
        """构建Chroma协议帧: HEAD(2) + ADDR(1) + CMD(1) + LEN(2) + DATA(N) + CHECKSUM(1)"""
        length = len(data)
        frame = struct.pack(">HBBH", self.HEAD, self.ADDR, cmd, length) + data
        checksum = sum(frame) & 0xFF
        return frame + struct.pack("B", checksum)

    def _parse_response(self, raw: bytes) -> bytes:
        """解析响应帧，校验checksum并返回DATA段"""
        if len(raw) < 7:
            raise ValueError(f"响应帧过短: {len(raw)} bytes")
        head, addr, cmd, length = struct.unpack(">HBBH", raw[:6])
        if head != self.HEAD:
            raise ValueError(f"帧头错误: 0x{head:04X}")
        data = raw[6:6+length]
        checksum = raw[6+length]
        expected = sum(raw[:6+length]) & 0xFF
        if checksum != expected:
            raise ValueError(f"Checksum校验失败: 0x{checksum:02X} != 0x{expected:02X}")
        return data

    def _send_cmd(self, cmd: int, data: bytes = b"") -> bytes:
        frame = self._build_frame(cmd, data)
        if self._simulation_mode:
            logger.debug(f"[SIM] Chroma TX: {frame.hex()}")
            # 仿真返回空数据帧
            resp_frame = self._build_frame(cmd, b"")
            return self._parse_response(resp_frame)
        self.send(frame)
        raw = self._recv_exact(7)  # 先读固定头
        head, addr, rcmd, length = struct.unpack(">HBBH", raw)
        if length > 0:
            raw += self._recv_exact(length + 1)  # DATA + CHECKSUM
        else:
            raw += self._recv_exact(1)  # CHECKSUM
        return self._parse_response(raw)

    def identify(self) -> str:
        data = self._send_cmd(self.CMD_IDN)
        if self._simulation_mode:
            return "SIM,Chroma63004,0,1.0"
        return data.decode("ascii").strip()

    def reset(self):
        self._send_cmd(self.CMD_RESET)

    def set_current(self, current_a: float):
        """设置CC模式电流(A)"""
        data = struct.pack(">f", current_a)
        self._send_cmd(self.CMD_SET_CURRENT, data)

    def measure(self) -> dict:
        """测量电压/电流/功率"""
        data = self._send_cmd(self.CMD_MEASURE)
        if self._simulation_mode:
            return {"voltage": 12.0, "current": 1.0, "power": 12.0}
        v, i, p = struct.unpack(">fff", data)
        return {"voltage": v, "current": i, "power": p}

8.2 事件总线（EventBus）

评审优化-P1：采用有界队列实现反压机制。非关键事件（如日志、状态上报）队列满时允许丢弃；关键控制事件（如资源释放、步骤完成）队列满时阻塞发布者，防止系统雪崩。

import asyncio
from collections import deque

class EventBus:
    def __init__(self, maxsize: int = 1000):
        self.subscribers: Dict[str, list] = {}
        self.critical_queue = asyncio.Queue(maxsize=maxsize)
        self.normal_queue = asyncio.Queue(maxsize=maxsize)

    async def publish(self, event_type: str, payload: dict, critical: bool = False):
        queue = self.critical_queue if critical else self.normal_queue
        if queue.full() and not critical:
            return  # 非关键事件丢弃
        await queue.put((event_type, payload))

    async def subscribe(self, event_type: str, callback):
        self.subscribers.setdefault(event_type, []).append(callback)

    async def dispatch(self):
        while True:
            try:
                event_type, payload = self.critical_queue.get_nowait()
            except asyncio.QueueEmpty:
                try:
                    event_type, payload = self.normal_queue.get_nowait()
                except asyncio.QueueEmpty:
                    await asyncio.sleep(0.005)
                    continue
            
            for cb in self.subscribers.get(event_type, []):
                await cb(payload)

8.6 调度器主循环

async def trigger_dependency_eval(self, step_id: str):
        for dep_id in self.step_registry.get_dependents(step_id):
            await self.try_submit(dep_id)

    async def try_submit(self, step_id: str):
        if self._can_schedule(step_id):
            asyncio.create_task(self.execute_step(step_id))

    async def execute_step(self, step_id: str):
        step = self.step_registry.get(step_id)
        await self.resource_manager.acquire(step.resources, step_id)
        self.step_registry.update_status(step_id, "RUNNING")
        try:
            result = await self.process_pool.run_script(step.script, step.context)
            self.step_registry.update_status(step_id, "COMPLETED", result)
            await self.event_bus.publish("STEP_COMPLETED", {"step_id": step_id}, critical=True)
        except Exception as e:
            await self._handle_failure(step_id, e)
        finally:
            await self.resource_manager.release(step.resources, step_id)

    async def on_step_status_changed(self, payload):
        await self.trigger_dependency_eval(payload["step_id"])

    async def periodic_scan(self):
        """使用优先队列处理超时和定时事件"""
        now = asyncio.get_event_loop().time()
        while self.step_registry.wake_heap and self.step_registry.wake_heap[0][0] <= now:
            _, step_id = heapq.heappop(self.step_registry.wake_heap)
            await self.try_submit(step_id)

class SequenceCompiler:
    def compile(self, sequence: list) -> list:
        compiled = []
        for node in sequence:
            if "loop" in node:
                compiled.extend(self._compile_loop(node))
            elif "branch" in node:
                compiled.extend(self._compile_branch(node))
            else:
                compiled.append(self._compile_step(node))
        return compiled

    def _compile_step(self, node: dict) -> dict:
        return {"type": "action", "id": node["id"], "script": node["script"]}

    def _compile_loop(self, node: dict) -> list:
        steps = []
        for i in range(node["loop"]["count"]):
            for sub in node["loop"]["sequence"]:
                step = self._compile_step(sub)
                step["id"] = f"{step['id']}_iter_{i}"
                steps.append(step)
        return steps

    def _compile_branch(self, node: dict) -> list:
        # 简化示例：实际需生成条件评估步骤
        return [self._make_action_step(node["branch"]["if_true"])]

    def _make_action_step(self, script: str) -> dict:
        return {"type": "action", "id": f"branch_{id(script)}", "script": script}

async def _handle_failure(self, step_id: str, error: Exception):
        step = self.step_registry.get(step_id)
        policy = step.failure_policy  # stop / skip / ignore
        
        if policy == "stop":
            self._running = False
            self.step_registry.update_status(step_id, "FAILED", str(error))
        elif policy == "skip":
            self.step_registry.update_status(step_id, "SKIPPED")

class DryRunner:
    """L4 序列逻辑仿真器，用于验证 DAG 依赖、变量流转与资源冲突"""
    
    def __init__(self, sequence_dag: dict):
        self.dag = sequence_dag
        self.steps = sequence_dag.get("steps", {})
        self.variables = sequence_dag.get("variables", {})
        
    def run(self) -> dict:
        """执行 DryRun 并返回验证报告"""
        report = {
            "valid": True,
            "errors": [],
            "warnings": [],
            "estimated_duration_ms": 0
        }
        
        # 1. 检查循环引用
        if self._has_circular_dependency():
            report["valid"] = False
            report["errors"].append("Circular dependency detected in sequence DAG")
            return report
            
        # 2. 拓扑排序并检查依赖存在性
        try:
            sorted_steps = self._topological_sort()
        except ValueError as e:
            report["valid"] = False
            report["errors"].append(str(e))
            return report
            
        # 3. 模拟变量传播与条件分支
        active_vars = set(self.variables.keys())
        for step_id in sorted_steps:
            step = self.steps[step_id]
            # 检查前置依赖变量是否已定义
            for dep in step.get("depends_on_vars", []):
                if dep not in active_vars:
                    report["valid"] = False
                    report["errors"].append(
                        f"Step '{step_id}' depends on undefined variable '{dep}'"
                    )
            # 模拟当前步骤输出变量
            for out_var in step.get("export_outputs", []):
                active_vars.add(out_var)
                
        # 4. 检查资源冲突
        resource_usage: Dict[str, List[str]] = defaultdict(list)
        for step_id, step in self.steps.items():
            for res in step.get("resources", []):
                resource_usage[res].append(step_id)
                
        for res, users in resource_usage.items():
            if len(users) > 1 and not self._has_mutex_switch(res, users):
                report["warnings"].append(
                    f"Resource '{res}' used by multiple steps without matrix switch: {users}"
                )
                
        report["estimated_duration_ms"] = self._estimate_duration(sorted_steps)
        return report
        
    def _has_circular_dependency(self) -> bool:
        """检测 DAG 中是否存在循环依赖"""
        visited = set()
        rec_stack = set()
        
        def dfs(node: str) -> bool:
            visited.add(node)
            rec_stack.add(node)
            for dep in self.steps.get(node, {}).get("depends_on", []):
                if dep not in visited:
                    if dfs(dep):
                        return True
                elif dep in rec_stack:
                    return True
            rec_stack.discard(node)
            return False
            
        for step_id in self.steps:
            if step_id not in visited:
                if dfs(step_id):
                    return True
        return False
        
    def _topological_sort(self) -> List[str]:
        """Kahn 算法拓扑排序"""
        in_degree = defaultdict(int)
        graph = defaultdict(list)
        all_nodes = set(self.steps.keys())
        
        for step_id, step in self.steps.items():
            for dep in step.get("depends_on", []):
                graph[dep].append(step_id)
                in_degree[step_id] += 1
                
        queue = deque([n for n in all_nodes if in_degree[n] == 0])
        sorted_nodes = []
        
        while queue:
            node = queue.popleft()
            sorted_nodes.append(node)
            for neighbor in graph[node]:
                in_degree[neighbor] -= 1
                if in_degree[neighbor] == 0:
                    queue.append(neighbor)
                    
        if len(sorted_nodes) != len(all_nodes):
            raise ValueError("Topological sort failed: graph has cycles")
        return sorted_nodes
        
    def _estimate_duration(self, sorted_steps: List[str]) -> int:
        """估算执行总时长（毫秒）"""
        total = 0
        for step_id in sorted_steps:
            total += self.steps[step_id].get("estimated_ms", 100)
        return total

import time
import random
import logging
from typing import Any, Dict, List, Optional
from dataclasses import dataclass, field

logger = logging.getLogger(__name__)

@dataclass
class FaultConfig:
    """故障注入配置"""
    fault_type: str          # timeout, drift, out_of_range, mode_fail
    probability: float = 0.0 # 触发概率 0.0~1.0
    count_limit: int = -1    # 最大触发次数，-1为无限
    params: Dict[str, Any] = field(default_factory=dict)

class MockInstrumentBase:
    """虚拟仪器基类，提供状态机、延迟模拟、故障注入与调用历史"""
    
    def __init__(self, instrument_id: str, config: Dict[str, Any]):
        self.instrument_id = instrument_id
        self.config = config
        self.state = "IDLE"
        self.call_history: List[Dict[str, Any]] = []
        self.faults: List[FaultConfig] = []
        self._fault_counters: Dict[str, int] = {}
        
        # 延迟模拟参数（高斯分布）
        self.delay_mean_ms = config.get("delay_mean_ms", 10)
        self.delay_std_ms = config.get("delay_std_ms", 2)
        
    def add_fault(self, fault: FaultConfig):
        """动态添加故障注入规则"""
        self.faults.append(fault)
        self._fault_counters[fault.fault_type] = 0
        
    def remove_fault(self, fault_type: str):
        """移除指定类型故障"""
        self.faults = [f for f in self.faults if f.fault_type != fault_type]
        
    def _simulate_delay(self):
        """模拟仪器响应延迟（高斯分布）"""
        delay_ms = max(0, random.gauss(self.delay_mean_ms, self.delay_std_ms))
        time.sleep(delay_ms / 1000.0)
        
    def _check_fault(self, action: str) -> Optional[Exception]:
        """检查是否触发故障注入"""
        for fault in self.faults:
            count = self._fault_counters.get(fault.fault_type, 0)
            if fault.count_limit > 0 and count >= fault.count_limit:
                continue
            if random.random() < fault.probability:
                self._fault_counters[fault.fault_type] = count + 1
                logger.warning(
                    f"[{self.instrument_id}] Fault injected: {fault.fault_type} "
                    f"on action '{action}'"
                )
                return self._create_fault_exception(fault, action)
        return None
        
    def _create_fault_exception(self, fault: FaultConfig, action: str) -> Exception:
        """根据故障类型创建对应异常"""
        msg = f"Mock fault [{fault.fault_type}] on {self.instrument_id}.{action}"
        if fault.fault_type == "timeout":
            return TimeoutError(msg)
        elif fault.fault_type == "connection_error":
            return ConnectionError(msg)
        elif fault.fault_type == "mode_fail":
            return RuntimeError(msg)
        return Exception(msg)
        
    def _record_call(self, action: str, params: Any, result: Any, error: Any = None):
        """记录调用历史"""
        self.call_history.append({
            "timestamp": time.time(),
            "action": action,
            "params": params,
            "result": result,
            "error": str(error) if error else None,
            "state": self.state
        })
        
    def get_call_history(self) -> List[Dict[str, Any]]:
        """获取调用历史记录"""
        return list(self.call_history)
        
    def clear_history(self):
        """清空调用历史"""
        self.call_history.clear()

from __future__ import annotations
import time
import random
from dataclasses import dataclass, field
from typing import Any, Callable, Dict, List, Optional
from enum import Enum


class TriggerType(Enum):
    COUNT = "count"
    PROBABILITY = "probability"
    TIME = "time"
    STATE = "state"


@dataclass
class FaultEvent:
    rule_id: str
    fault_kind: str
    message: str
    timestamp: float = field(default_factory=time.time)
    metadata: Dict[str, Any] = field(default_factory=dict)


class FaultRule:
    """单条故障注入规则"""

    def __init__(self, config: Dict[str, Any]):
        self.id: str = config["id"]
        self.type: TriggerType = TriggerType(config["type"])
        self.target: str = config["target"]
        self.trigger_cfg: Dict[str, Any] = config["trigger"]
        self.fault_cfg: Dict[str, Any] = config["fault"]
        self._call_count: int = 0
        self._start_time: float = time.time()
        self._triggered: bool = False

    def matches(self, target: str) -> bool:
        """检查目标是否匹配本规则"""
        return self.target == target

    def trigger(self, state_evaluator: Optional[Callable[[str], Any]] = None) -> bool:
        """评估是否应触发故障"""
        if self._triggered and self.type == TriggerType.COUNT:
            if self.trigger_cfg.get("exact"):
                return False

        if self.type == TriggerType.COUNT:
            self._call_count += 1
            expected = self.trigger_cfg["count"]
            if self.trigger_cfg.get("exact"):
                self._triggered = self._call_count == expected
            else:
                self._triggered = self._call_count >= expected

        elif self.type == TriggerType.PROBABILITY:
            prob = self.trigger_cfg["probability"]
            self._triggered = random.random() < prob

        elif self.type == TriggerType.TIME:
            elapsed = time.time() - self._start_time
            after = self.trigger_cfg["after_seconds"]
            window = self.trigger_cfg.get("window_seconds", 1)
            self._triggered = after <= elapsed <= after + window

        elif self.type == TriggerType.STATE:
            if state_evaluator is None:
                raise ValueError("State trigger requires state_evaluator")
            condition = self.trigger_cfg["condition"]
            # 简单表达式求值（生产环境应使用安全沙箱）
            self._triggered = bool(eval(condition, {"__builtins__": {}}, 
                                         {"state": state_evaluator}))

        return self._triggered

    def create_fault_event(self) -> FaultEvent:
        """生成故障事件"""
        return FaultEvent(
            rule_id=self.id,
            fault_kind=self.fault_cfg["kind"],
            message=self.fault_cfg["message"],
            metadata={k: v for k, v in self.fault_cfg.items() 
                      if k not in ("kind", "message")}
        )


class FaultInjectionEngine:
    """故障注入引擎"""

    def __init__(self):
        self._rules: Dict[str, FaultRule] = {}
        self._events: List[FaultEvent] = []
        self._state_evaluator: Optional[Callable[[str], Any]] = None

    def load_rules(self, rules_config: List[Dict[str, Any]]) -> None:
        for cfg in rules_config:
            rule = FaultRule(cfg)
            self._rules[rule.id] = rule

    def set_state_evaluator(self, evaluator: Callable[[str], Any]) -> None:
        self._state_evaluator = evaluator

    def evaluate(self, target: str) -> Optional[FaultEvent]:
        """评估目标是否命中故障规则"""
        for rule in self._rules.values():
            if rule.matches(target):
                if rule.trigger(self._state_evaluator):
                    event = rule.create_fault_event()
                    self._events.append(event)
                    return event
        return None

    def get_report(self) -> Dict[str, Any]:
        """生成故障注入报告"""
        return {
            "total_rules": len(self._rules),
            "total_events": len(self._events),
            "events": [
                {
                    "rule_id": e.rule_id,
                    "fault_kind": e.fault_kind,
                    "message": e.message,
                    "timestamp": e.timestamp,
                }
                for e in self._events
            ],
        }

    def reset(self) -> None:
        self._events.clear()
        for rule in self._rules.values():
            rule._call_count = 0
            rule._start_time = time.time()
            rule._triggered = False

class CallRecorder:
    """线程安全的仪器调用录制器"""

    FLUSH_THRESHOLD = 50

    def __init__(self, filepath: Path, execution_id: str, uut_id: str):
        self._filepath = filepath
        self._execution_id = execution_id
        self._uut_id = uut_id
        self._lock = threading.Lock()
        self._buffer: List[Dict[str, Any]] = []
        self._seq = 0
        self._file = None
        self._active = False

    def start(self) -> None:
        with self._lock:
            self._file = open(self._filepath, "w", encoding="utf-8")
            header = {
                "_meta": True,
                "type": "header",
                "execution_id": self._execution_id,
                "uut_id": self._uut_id,
                "start_time": time.strftime("%Y-%m-%dT%H:%M:%SZ", time.gmtime()),
                "version": "3.2",
            }
            self._file.write(json.dumps(header, ensure_ascii=False) + "\n")
            self._active = True

    def record(self, resource_id: str, action: str, command: str,
               response: str, elapsed_ms: int, step_id: str = "",
               error: Optional[str] = None, protocol: str = "SCPI") -> None:
        if not self._active:
            return
        with self._lock:
            self._seq += 1
            rec = CallRecord(
                seq=self._seq,
                timestamp=time.strftime("%Y-%m-%dT%H:%M:%S.", time.gmtime()) 
                          + f"{int(time.time()*1000)%1000:03d}Z",
                execution_id=self._execution_id,
                uut_id=self._uut_id,
                step_id=step_id,
                resource_id=resource_id,
                action=action,
                command=command,
                response=response,
                elapsed_ms=elapsed_ms,
                error=error,
                protocol=protocol,
            )
            self._buffer.append(asdict(rec))
            self._flush_locked()

    def stop(self) -> None:
        with self._lock:
            self._active = False
            self._flush_locked(force=True)
            if self._file:
                footer = {
                    "_meta": True,
                    "type": "footer",
                    "end_time": time.strftime("%Y-%m-%dT%H:%M:%SZ", time.gmtime()),
                    "total_records": self._seq,
                }
                self._file.write(json.dumps(footer, ensure_ascii=False) + "\n")
                self._file.close()
                self._file = None

    def _flush_locked(self, force: bool = False) -> None:
        """必须在持有锁的情况下调用"""
        if not self._file:
            return
        if force or len(self._buffer) >= self.FLUSH_THRESHOLD:
            for rec in self._buffer:
                self._file.write(json.dumps(rec, ensure_ascii=False) + "\n")
            self._file.flush()
            self._buffer.clear()

from __future__ import annotations
import json
import time
from typing import Any, Dict, List, Optional, Tuple
from pathlib import Path


class ReplayEngine:
    """录制回放引擎"""

    def __init__(self, filepath: Path, speed_factor: float = 1.0,
                 enable_timing: bool = False):
        self._filepath = filepath
        self._speed_factor = max(0.01, speed_factor)
        self._enable_timing = enable_timing
        self._records: Dict[Tuple[str, str, str], List[Dict[str, Any]]] = {}
        self._call_counters: Dict[Tuple[str, str, str], int] = {}
        self._last_timestamp: Optional[float] = None
        self._loaded = False

    def _load(self) -> None:
        if self._loaded:
            return
        with open(self._filepath, "r", encoding="utf-8") as f:
            for line in f:
                line = line.strip()
                if not line:
                    continue
                rec = json.loads(line)
                if rec.get("_meta"):
                    continue
                key = (rec["resource_id"], rec["action"], rec["command"])
                self._records.setdefault(key, []).append(rec)
        self._loaded = True

    def start(self) -> None:
        self._load()
        self._call_counters.clear()
        self._last_timestamp = None

    def get_response(self, resource_id: str, action: str, 
                     command: str) -> Dict[str, Any]:
        key = (resource_id, action, command)
        idx = self._call_counters.get(key, 0)
        records = self._records.get(key, [])

        if idx >= len(records):
            return {
                "response": "",
                "error": f"No matching recording for {key} at index {idx}",
                "elapsed_ms": 0,
            }

        rec = records[idx]
        self._call_counters[key] = idx + 1

        # 可选：模拟时间间隔
        if self._enable_timing and self._last_timestamp is not None:
            rec_ts = time.mktime(time.strptime(
                rec["timestamp"].replace("Z", ""), "%Y-%m-%dT%H:%M:%S.%f"))
            delta = (rec_ts - self._last_timestamp) / self._speed_factor
            if delta > 0:
                time.sleep(delta)
        self._last_timestamp = time.time()

        return {
            "response": rec["response"],
            "error": rec.get("error"),
            "elapsed_ms": rec["elapsed_ms"],
        }

    expected: Any
    actual: Any
    detail: str = ""


class ReplayValidator:
    """回放结果校验器"""

    TIMING_TOLERANCE_MS = 500

    def compare(self, original_path: str, replay_events: List[Dict[str, Any]]
                ) -> List[DiffEntry]:
        diffs: List[DiffEntry] = []
        original_records = self._load_records(original_path)

        # 按key分组统计
        orig_counts: Dict[str, int] = {}
        replay_counts: Dict[str, int] = {}
        for rec in original_records:
            k = f"{rec['resource_id']}|{rec['action']}|{rec['command']}"
            orig_counts[k] = orig_counts.get(k, 0) + 1
        for evt in replay_events:
            k = f"{evt['resource_id']}|{evt['action']}|{evt['command']}"
            replay_counts[k] = replay_counts.get(k, 0) + 1

        # 检查调用次数
        all_keys = set(orig_counts) | set(replay_counts)
        for k in all_keys:
            oc = orig_counts.get(k, 0)
            rc = replay_counts.get(k, 0)
            if oc != rc:
                parts = k.split("|")
                diffs.append(DiffEntry(
                    diff_type="call_count_mismatch",
                    resource_id=parts[0],
                    command=parts[2],
                    expected=oc, actual=rc,
                ))

        # 检查响应值差异
        orig_map = {}
        for rec in original_records:
            k = (rec["resource_id"], rec["action"], rec["command"])
            orig_map.setdefault(k, []).append(rec["response"])

        replay_map: Dict[tuple, List[str]] = {}
        for evt in replay_events:
            k = (evt["resource_id"], evt["action"], evt["command"])
            replay_map.setdefault(k, []).append(evt.get("response", ""))

        for k in orig_map:
            if k in replay_map:
                for i, (ov, rv) in enumerate(
                        zip(orig_map[k], replay_map[k])):
                    if ov != rv:
                        diffs.append(DiffEntry(
                            diff_type="value_difference",
                            resource_id=k[0], command=k[2],
                            expected=ov, actual=rv,
                            detail=f"index={i}",
                        ))

        return diffs

    def _load_records(self, path: str) -> List[Dict[str, Any]]:
        import json
        records = []
        with open(path, "r") as f:
            for line in f:
                rec = json.loads(line.strip())
                if not rec.get("_meta"):
                    records.append(rec)
        return records

// composables/useSimulationStream.ts
import { ref, onUnmounted } from 'vue';
import type { SimulationState, LogEntry } from '@/types/simulation';

interface UseSimulationStreamOptions {
  onEvent: (event: string, data: any) => void;
  onError?: (error: Event) => void;
}

export function useSimulationStream(options: UseSimulationStreamOptions) {
  const connected = ref(false);
  const lastEventId = ref<string | null>(null);
  let eventSource: EventSource | null = null;
  let reconnectAttempts = 0;
  let reconnectTimer: ReturnType<typeof setTimeout> | null = null;

  // 指数退避重连：初始1s，最大30s
  const BASE_DELAY = 1000;
  const MAX_DELAY = 30000;

  function getReconnectDelay(): number {
    const delay = Math.min(BASE_DELAY * Math.pow(2, reconnectAttempts), MAX_DELAY);
    return delay;
  }

  function connect(simulationId: string) {
    disconnect();

    const url = new URL(`/api/v1/simulation/${simulationId}/stream`, window.location.origin);
    // 携带最后接收的事件ID，请求增量数据
    if (lastEventId.value) {
      url.searchParams.set('last_event_id', lastEventId.value);
    }

    eventSource = new EventSource(url.toString());

    eventSource.onopen = () => {
      connected.value = true;
      reconnectAttempts = 0; // 重连成功，重置计数器
    };

    eventSource.onmessage = (event) => {
      if (event.lastEventId) {
        lastEventId.value = event.lastEventId;
      }
      try {
        const parsed = JSON.parse(event.data);
        options.onEvent(parsed.type, parsed.data);
      } catch (e) {
        console.error('SSE parse error:', e);
      }
    };

    eventSource.onerror = (error) => {
      connected.value = false;
      eventSource?.close();
      eventSource = null;

      // 指数退避重连
      const delay = getReconnectDelay();
      reconnectAttempts++;
      console.warn(`SSE connection lost. Reconnecting in {delay}ms (attempt {reconnectAttempts})...`);

      reconnectTimer = setTimeout(() => {
        connect(simulationId);
      }, delay);

      options.onError?.(error);
    };
  }

  function disconnect() {
    if (reconnectTimer) {
      clearTimeout(reconnectTimer);
      reconnectTimer = null;
    }
    if (eventSource) {
      eventSource.close();
      eventSource = null;
    }
    connected.value = false;
    reconnectAttempts = 0;
  }

  onUnmounted(() => {
    disconnect();
  });

  return { connected, lastEventId, connect, disconnect };
}

class FixtureState:

    actuator_states: Dict[str, str] = field(default_factory=dict)


class FixtureController:
    """夹具控制器，管理夹具的夹紧/释放、路由切换、传感器读取等操作。"""

    def __init__(self, fixture_id: str, io_driver: Any):
        self.fixture_id = fixture_id
        self._io = io_driver
        self._state = FixtureState()

    def clamp(self, timeout_ms: int = 5000) -> bool:
        """夹紧夹具，返回是否成功。"""
        success = self._io.write(f"FIX:{self.fixture_id}:CLAMP", timeout_ms=timeout_ms)
        if success:
            self._state.clamped = True
        return success

    def release(self, timeout_ms: int = 5000) -> bool:
        """释放夹具，返回是否成功。"""
        success = self._io.write(f"FIX:{self.fixture_id}:RELEASE", timeout_ms=timeout_ms)
        if success:
            self._state.clamped = False
        return success

    def set_route(self, route_id: str, close: bool = True) -> bool:
        """设置继电器路由路径。"""
        action = "CLOSE" if close else "OPEN"
        success = self._io.write(
            f"FIX:{self.fixture_id}:ROUTE:{route_id}:{action}"
        )
        if success:
            self._state.relay_states[route_id] = "closed" if close else "open"
        return success

    def read_sensor(self, sensor_id: str) -> Optional[float]:
        """读取指定传感器值。"""
        value = self._io.query(f"FIX:{self.fixture_id}:SENS:{sensor_id}?")
        if value is not None:
            try:
                float_val = float(value)
                self._state.sensor_values[sensor_id] = float_val
                return float_val
            except ValueError:
                return None
        return None

    def get_state(self) -> FixtureState:
        """获取夹具当前完整状态快照。"""
        # 刷新所有传感器
        for sensor_id in list(self._state.sensor_values.keys()):
            self.read_sensor(sensor_id)
        return self._state

16.1 设计原则

操作员面板是产线人员与测试系统交互的核心界面，其设计直接关系到测试效率与操作准确性。本系统遵循以下五大核心设计原则：

大字体、高对比度：充分考虑产线环境的光线条件与视距需求，确保关键信息在远距离下依然清晰可读。

操作极简：将核心工作流精简为“扫码 → 开始 → 看结果”最多3步，降低认知负荷与操作门槛。

状态醒目：测试结果采用大面积颜色区分（通过/失败），确保操作员无需仔细阅读即可瞬间获取结果状态。

防误操作：对关键操作（如停止测试、修改配置等）强制进行二次确认，防止因误触导致的生产事故。

异常引导：当测试失败时，系统不仅展示错误代码，还会提供明确的下一步操作指引与诊断建议，缩短异常处理时间。

class NATSClient:
    def __init__(self, station_id: str, nats_url: str):
        self.station_id = station_id
        self.nats_url = nats_url
        self._client: Optional[NatsClient] = None
        self._js: Optional[JetStreamContext] = None

    async def connect(self):
        self._client = await nats.connect(self.nats_url)
        self._js = self._client.jetstream()

        return f"teststation.{self.station_id}.{category}.{action}"

    async def send_command(self, action: str, payload: Dict[str, Any]):
        topic = self._topic("cmd", action)
        await self._client.publish(topic, json.dumps(payload).encode())

    async def publish_event(self, action: str, payload: Dict[str, Any]):
        topic = self._topic("event", action)
        await self._client.publish(topic, json.dumps(payload).encode())

    async def subscribe(self, category: str, action: str, handler: Callable):
        topic = self._topic(category, action)
        await self._client.subscribe(topic, cb=handler)

    async def request(self, category: str, action: str, payload: Dict[str, Any], timeout: float = 5.0):
        topic = self._topic("req", action)
        response = await self._client.request(topic, json.dumps(payload).encode(), timeout=timeout)
        return json.loads(response.data.decode())

        self._rules[rule.name] = rule

        rule = self._rules.get(metric_name)
        if not rule:

        else:
            self._states[metric_name] = 0

十六、操作员面板

16.1 设计原则

操作员面板是产线人员与测试系统交互的核心界面，其设计直接关系到测试效率与操作准确性。本系统遵循以下五大核心设计原则：

大字体、高对比度：充分考虑产线环境的光线条件与视距需求，确保关键信息在远距离下依然清晰可读。

操作极简：将核心工作流精简为“扫码 → 开始 → 看结果”最多3步，降低认知负荷与操作门槛。

状态醒目：测试结果采用大面积颜色区分（通过/失败），确保操作员无需仔细阅读即可瞬间获取结果状态。

防误操作：对关键操作（如停止测试、修改配置等）强制进行二次确认，防止因误触导致的生产事故。

异常引导：当测试失败时，系统不仅展示错误代码，还会提供明确的下一步操作指引与诊断建议，缩短异常处理时间。
